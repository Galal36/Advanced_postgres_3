CREATE FUNCTION is_odd_or_even(num INT) RETURNS VARCHAR AS $$
BEGIN
    IF num % 2 = 0 THEN
        RETURN 'Even';
    ELSE
        RETURN 'Odd';
    END IF;
END;
$$ LANGUAGE plpgsql;

    id INT,
    e_name VARCHAR,
    email VARCHAR,
    address VARCHAR,
    track_id INT,
    birth_date DATE,
    gender VARCHAR
) AS $$
BEGIN
    RETURN QUERY SELECT * FROM student WHERE id = student_id;
END;
$$ LANGUAGE plpgsql;


--Q5
CREATE FUNCTION avg_grade_for_subject(subject_name VARCHAR) RETURNS FLOAT AS $$
BEGIN
    RETURN (SELECT AVG(g.grade)
            FROM grades g
            JOIN subject s ON g.sub_id = s.id
            WHERE s.sub_name = subject_name);
END;
$$ LANGUAGE plpgsql;

--Q7
CREATE TABLE Student_Changes (
    id SERIAL PRIMARY KEY,
    student_id INT,
    action VARCHAR,
    action_time TIMESTAMP,
    description TEXT
);


--then create the trigger function:


CREATE FUNCTION monitor_student_changes() RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO Student_Changes (student_id, action, action_time, description)
        VALUES (NEW.id, 'INSERT', NOW(), 'New student added');
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO Student_Changes (student_id, action, action_time, description)
        VALUES (NEW.id, 'UPDATE', NOW(), 'Student record updated');
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO Student_Changes (student_id, action, action_time, description)
        VALUES (OLD.id, 'DELETE', NOW(), 'Student record deleted');
        RETURN OLD;
    END IF;
END;
$$ LANGUAGE plpgsql;


--finally create and execute the trigger


CREATE TRIGGER after_student_change
AFTER INSERT OR UPDATE OR DELETE ON student
FOR EACH ROW
EXECUTE FUNCTION monitor_student_changes();
